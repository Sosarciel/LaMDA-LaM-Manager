import type { MPromise } from "@zwa73/utils";

import type { AnyTextCompletionRequest } from "RequestFormat";
import type { AnyTextCompletionResponse } from "ResponseFormat";
import type { TokensizerType } from "Tokensizer";

import type { RespFormatter, TextCompletionTaskFormatter } from "Task/ToolInterface";

import type { InstructTaskOption } from "./Interface";

/**
 * Instruct 任务格式化工具
 *
 * 设计考虑：
 * 1. 与 ChatTaskFormatter 结构相似，但针对 Instruct 任务简化
 * 2. 不包含 chat-specific 方法（buildMessage, formatMessage）
 * 3. 添加 buildPrompt 方法处理不同 FIM 模式的提示构建
 * 4. computeTokenCount 接收 prompt 字符串，而不是消息列表
 * 5. 保持与 TextCompletionTaskFormatter 和 RespFormatter 的兼容性
 *
 * @typeParam REQ - 模型特定的请求格式，必须扩展 AnyTextCompletionRequest
 * @typeParam RES - 模型特定的响应格式，必须扩展 AnyTextCompletionResponse
 */
export type InstructTaskFormatter<
  REQ extends AnyTextCompletionRequest,
  RES extends AnyTextCompletionResponse
> = TextCompletionTaskFormatter<InstructTaskOption, REQ, RES> &
  RespFormatter<RES> & {
    /**
     * 构建模型特定的提示格式
     * @param opt - Instruct 任务选项
     * @returns 模型特定的提示字符串或对象
     *
     * 设计考虑：
     * - 统一处理各种 FIM 模式（prefix-only, prefix-suffix, prompt-only 等）
     * - 返回类型为 string，因为大多数 instruct 模型接受字符串提示
     * - 对于需要特殊格式的模型（如 JSON），可在具体格式化器中处理
     */
    buildPrompt(opt: InstructTaskOption): string;

    /**
     * 计算提示文本的 Token 数量
     * @param prompt - 待计算的提示文本
     * @param tokensizerType - Tokenizer 类型
     * @returns token 数
     *
     * 设计考虑：
     * - 参数简化为 prompt 字符串，因为 Instruct 任务不需要复杂的历史消息
     * - 实际实现需要考虑 FIM 模式下的特殊分隔符 token
     * - 使用模型的 tokensizerType 确保计算准确性
     */
    computeTokenCount(prompt: string, tokensizerType: TokensizerType): Promise<number>;

    // 注意：不包含以下 chat-specific 方法：
    // - buildMessage: Instruct 任务不需要消息构建
    // - formatMessage: Instruct 任务不需要消息格式化
  };

/**
 * Instruct 任务格式化器注册表
 *
 * 设计考虑：
 * 1. 与 ChatTaskFormaterTable 结构一致，保持系统统一
 * 2. 键名使用 model_type 格式，便于模型识别
 * 3. 值是对应的格式化器实现
 * 4. 支持动态扩展，新格式化器只需添加到表中
 *
 * 当前支持的模式：
 * - openai_instruct: OpenAI Instruct 模式（如 gpt-3.5-turbo-instruct）
 * - deepseek_fim: DeepSeek FIM 模式（填充中间）
 * - prefix_completion: 通用前缀续写模式
 */
export const InstructTaskFormaterTable = {
  // 将在后续实现中导入并添加具体格式化器
  // openai_instruct: OpenAIInstructFormatter,
  // deepseek_fim: DeepseekFIMFormatter,
  // prefix_completion: PrefixCompletionFormatter,
} as const;

/**
 * Instruct 格式化器类型
 *
 * 设计考虑：
 * 1. 从注册表自动推导所有可用的格式化器类型
 * 2. 提供类型安全，确保只有注册的格式化器可以被使用
 * 3. 便于 TypeScript 类型检查和自动补全
 */
export type InstructFormaterType = keyof typeof InstructTaskFormaterTable;

/**
 * 获取指定模型的 Instruct 格式化器
 *
 * 设计考虑：
 * 1. 提供类型安全的格式化器获取函数
 * 2. 支持模型 ID 到格式化器类型的映射
 * 3. 处理未知模型的降级策略
 *
 * @param modelId - 模型 ID（如 "gpt-3.5-turbo-instruct", "deepseek-chat"）
 * @returns 对应的格式化器类型，如果未找到则返回 undefined
 *
 * 示例映射：
 * - "gpt-3.5-turbo-instruct" -> "openai_instruct"
 * - "deepseek-chat" -> "deepseek_fim"（如果支持 FIM）
 * - "deepseek-coder" -> "deepseek_fim"
 */
export function getInstructFormaterType(modelId: string): InstructFormaterType | undefined {
  // 模型 ID 到格式化器类型的映射逻辑
  // 注意：这是一个简化版本，实际实现可能需要更复杂的匹配逻辑
  if (modelId.includes('gpt-3.5-turbo-instruct') || modelId.includes('davinci')) {
    return 'openai_instruct' as InstructFormaterType;
  } else if (modelId.includes('deepseek')) {
    return 'deepseek_fim' as InstructFormaterType;
  } else {
    // 默认返回通用前缀续写格式化器
    return 'prefix_completion' as InstructFormaterType;
  }
}

/**
 * Instruct 任务适配器工具函数
 *
 * 设计考虑：
 * 1. 提供常用工具函数，减少格式化器重复代码
 * 2. 统一处理通用逻辑，如 FIM 模式检测、提示组合等
 */
export const InstructAdapterUtils = {
  /**
   * 组合 FIM 提示
   *
   * 设计考虑：
   * 1. 统一处理不同 FIM 模式的提示组合逻辑
   * 2. 提供可配置的分隔符，适应不同模型的要求
   * 3. 处理空值情况，避免不必要的分隔符
   *
   * @param prefix - 前缀文本（可选）
   * @param suffix - 后缀文本（可选）
   * @param prompt - 主要提示文本（可选，当不作为 prefix/suffix 时）
   * @param options - 配置选项
   * @returns 组合后的提示字符串
   */
  combineFIMPrompt(
    prefix?: string,
    suffix?: string,
    prompt?: string,
    options?: {
      prefixSuffixSeparator?: string; // prefix 和 suffix 之间的分隔符
      prefixPromptSeparator?: string; // prefix 和 prompt 之间的分隔符
      promptSuffixSeparator?: string; // prompt 和 suffix 之间的分隔符
    }
  ): string {
    const {
      prefixSuffixSeparator = '', // DeepSeek FIM 使用特殊 token，这里为空
      prefixPromptSeparator = '\n', // prefix 和 prompt 之间通常换行
      promptSuffixSeparator = '\n', // prompt 和 suffix 之间通常换行
    } = options || {};

    const hasPrefix = prefix !== undefined && prefix.trim() !== '';
    const hasSuffix = suffix !== undefined && suffix.trim() !== '';
    const hasPrompt = prompt !== undefined && prompt.trim() !== '';

    if (hasPrefix && hasSuffix) {
      // 完整 FIM 模式：prefix + suffix（中间部分由模型生成）
      return `${prefix}${prefixSuffixSeparator}${suffix}`;
    } else if (hasPrefix && hasPrompt) {
      // prefix + prompt 模式：将 prompt 作为 suffix 的替代
      return `${prefix}${prefixPromptSeparator}${prompt}`;
    } else if (hasPrompt && hasSuffix) {
      // prompt + suffix 模式：将 prompt 作为 prefix 的替代
      return `${prompt}${promptSuffixSeparator}${suffix}`;
    } else if (hasPrefix) {
      // 仅 prefix：前缀续写模式
      return prefix;
    } else if (hasSuffix) {
      // 仅 suffix：后缀补全模式（不常见）
      return suffix;
    } else {
      // 仅 prompt：普通文本生成模式
      return prompt || '';
    }
  },

  /**
   * 验证 Instruct 任务选项
   *
   * 设计考虑：
   * 1. 提供统一的选项验证逻辑
   * 2. 早期发现问题，避免无效请求
   * 3. 返回详细的错误信息，便于调试
   *
   * @param opt - Instruct 任务选项
   * @returns 验证结果，如果无效则返回错误信息
   */
  validateInstructOption(opt: InstructTaskOption): { valid: true } | { valid: false; error: string } {
    // 检查必需字段
    if (!opt.prompt || opt.prompt.trim() === '') {
      return { valid: false, error: 'Instruct 任务必须提供非空的 prompt 字段' };
    }

    // 检查 FIM 模式参数
    const hasPrefix = opt.prefix !== undefined && opt.prefix.trim() !== '';
    const hasSuffix = opt.suffix !== undefined && opt.suffix.trim() !== '';

    if (hasPrefix && hasSuffix && opt.prompt) {
      // 完整 FIM 模式：三个字段都提供，可能产生歧义
      // 这里不报错，但记录警告（由具体格式化器决定如何处理）
    }

    // 检查 Token 限制
    if (opt.max_tokens !== undefined && opt.max_tokens <= 0) {
      return { valid: false, error: 'max_tokens 必须大于 0' };
    }

    // 检查温度参数范围
    if (opt.temperature !== undefined && (opt.temperature < 0 || opt.temperature > 2)) {
      return { valid: false, error: 'temperature 必须在 0 到 2 之间' };
    }

    // 检查 top_p 参数范围
    if (opt.top_p !== undefined && (opt.top_p <= 0 || opt.top_p > 1)) {
      return { valid: false, error: 'top_p 必须在 0（不含）到 1 之间' };
    }

    return { valid: true };
  },
};
